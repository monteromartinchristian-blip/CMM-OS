- [ ] **Step 2: Implement reference-only `to_authority_reference_dict()`**

Add to `PermissionGateResult`:

```python
def to_authority_reference_dict(self) -> dict[str, Any]:
    items = ()
    if self.approval_evidence:
        items = self.approval_evidence.get("items", ())

    approvals = []
    for item in items:
        approvals.append(
            {
                "requirement_id": item.get("requirement_id"),
                "action": item.get("action"),
                "approval_request_id": item.get("request_id"),
                "approval_decision_ids": sorted(
                    set(item.get("approval_decision_ids", ()))
                ),
            }
        )

    return {
        "permission_decision_id": self.decision_id,
        "outcome": self.outcome,
        "approvals": sorted(
            approvals,
            key=lambda value: (
                value.get("action") or "",
                value.get("requirement_id") or "",
                value.get("approval_request_id") or "",
            ),
        ),
    }
```

Use immutable/reference-safe JSON-compatible values.

Do not put actor, comments, approval parameters or permission reasoning into this payload.
